  <project_name>POS POC - 團購菜單與組合/加購系統</project_name>

  <overview>
    建立可被分享與 fork 的菜單系統，支援巢狀選項與多種優惠規則（組合價、加購、買一送一）。
    以團購為核心場景：每個參與者在單一團內有自己的購物車；提交後由團主彙整、可修改或駁回。
    價格規則可疊加；無庫存、單階段購物車。

    本規格專注於待完成的前端 UI 功能，特別是：
    - 編輯器系統（菜單、規則、Combo 編輯器）
    - AI 菜單生成功能
    - 團主操作介面
  </overview>

      <optimistic_update>
        <title>Optimistic Update 策略</title>
        <features>
          - 立即更新 UI（不等待伺服器回應）
          - 失敗時自動回滾
          - 顯示同步狀態（Saving... / Saved / Error）
          - 衝突偵測與解決
        </features>
        <implementation>
          ```typescript
          // 使用 SWR mutate 進行 optimistic update
          async function saveWithOptimistic&lt;T&gt;(
            key: string,
            data: T,
            saveFn: (data: T) => Promise&lt;T&gt;
          ) {
            // 1. 樂觀更新本地快取
            mutate(key, data, false);

            try {
              // 2. 實際儲存到伺服器
              const result = await saveFn(data);
              // 3. 用伺服器回應更新快取
              mutate(key, result, false);
              return { success: true, data: result };
            } catch (error) {
              // 4. 失敗時回滾
              mutate(key);
              return { success: false, error };
            }
          }
          ```
        </implementation>
      </optimistic_update>

          function handleDragEnd(event: DragEndEvent) {
            const { active, over } = event;
            if (!over || active.id === over.id) return;

            setItems(items => {
              const oldIndex = items.findIndex(i => i.id === active.id);
              const newIndex = items.findIndex(i => i.id === over.id);
              const reordered = arrayMove(items, oldIndex, newIndex);

              // 更新 display_order
              return reordered.map((item, index) => ({
                ...item,
                display_order: index
              }));

        <implementation>
          ```typescript
          // Server Action: duplicateMenuItem
          'use server'
          import { nanoid } from 'nanoid';

          export async function duplicateMenuItem(
            menuId: string,
            itemId: string
          ): Promise&lt;MenuItem&gt; {
            const supabase = createClient();

            // 1. 取得原品項及所有巢狀資料
            const { data: original } = await supabase
              .from('menu_item')
              .select(`
                *,
                option_groups:menu_item_option_group(
                  option_group:option_group(
                    *,
                    choices:option_choice(*)
                  )
                )
              `)
              .eq('id', itemId)
              .single();

            if (!original) throw new Error('Item not found');

            // 2. 建立新品項
            const newItem = await supabase
              .from('menu_item')
              .insert({
                ...omit(original, ['id', 'created_at', 'option_groups']),
                id: nanoid(),
                name: `${original.name} (複製)`,
                display_order: original.display_order + 1
              })
              .select()
              .single();

            // 3. 遞迴複製選項群組和選項
            for (const og of original.option_groups) {
              const newGroup = await duplicateOptionGroup(og.option_group);
              await linkItemToGroup(newItem.data.id, newGroup.id);
            }

            return newItem.data;
          }
          ```
        </implementation>

          // lib/price.ts

          /**
           * 將分轉換為元的顯示格式
           * @example formatPrice(3500) => "$35.00"
           */
          export function formatPrice(cents: number): string {
            return `$${(cents / 100).toFixed(2)}`;
          }

          /**
           * 將元轉換為分
           * @example parsePriceToCents("35.00") => 3500
           */
          export function parsePriceToCents(dollars: string): number {
            const num = parseFloat(dollars);
            if (isNaN(num)) return 0;
            return Math.round(num * 100);
          }

          /**
           * 格式化價格增減
           * @example formatPriceDelta(1500) => "+$15.00"
           * @example formatPriceDelta(-500) => "-$5.00"
           */
          export function formatPriceDelta(cents: number): string {
            const prefix = cents >= 0 ? '+' : '';
            return `${prefix}$${(cents / 100).toFixed(2)}`;
          }
          ```

          export function getOrCreateDeviceId(): string {
            // 1. 嘗試從 localStorage 讀取
            let deviceId = localStorage.getItem('anonymous_device_id');

            // 2. 若無，嘗試從 cookie 讀取
            if (!deviceId) {
              deviceId = getCookie('anonymous_device_id');
            }

            // 3. 若仍無，生成新的
            if (!deviceId) {
              deviceId = crypto.randomUUID();
            }

            // 4. 同步儲存到 localStorage 和 cookie
            localStorage.setItem('anonymous_device_id', deviceId);
            setCookie('anonymous_device_id', deviceId, { maxAge: 365 * 24 * 60 * 60 });

            return deviceId;
          }

          // hooks/useImageUpload.ts
          interface UploadProgress {
            fileId: string;
            fileName: string;
            progress: number;     // 0-100
            bytesUploaded: number;
            totalBytes: number;
            status: 'pending' | 'uploading' | 'processing' | 'complete' | 'error';
            error?: string;
          }

          export function useImageUpload(scope: MediaScope) {
            const [uploads, setUploads] = useState&lt;UploadProgress[]&gt;([]);

            async function uploadFile(file: File, targetId: string) {
              const fileId = crypto.randomUUID();

              // 初始化進度
              setUploads(prev => [...prev, {
                fileId,
                fileName: file.name,
                progress: 0,
                bytesUploaded: 0,
                totalBytes: file.size,
                status: 'pending'
              }]);

              try {
                // 1. 取得簽名 URL
                const { signedUrl, path } = await createSignedUploadUrl(scope, targetId);

                // 2. 使用 XHR 上傳（支援進度追蹤）
                await uploadWithProgress(signedUrl, file, (progress) => {
                  setUploads(prev => prev.map(u =>
                    u.fileId === fileId
                      ? { ...u, progress, bytesUploaded: (progress / 100) * file.size, status: 'uploading' }
                      : u
                  ));
                });

                // 3. 發布媒體
                setUploads(prev => prev.map(u =>
                  u.fileId === fileId ? { ...u, status: 'processing' } : u
                ));
                await publishMedia(path, scope, targetId);

                // 4. 完成
                setUploads(prev => prev.map(u =>
                  u.fileId === fileId ? { ...u, status: 'complete', progress: 100 } : u
                ));
              } catch (error) {
                setUploads(prev => prev.map(u =>
                  u.fileId === fileId ? { ...u, status: 'error', error: error.message } : u
                ));
              }
            }

            function cancelUpload(fileId: string) {
              // 取消上傳邏輯
            }

            return { uploads, uploadFile, cancelUpload };
          }
          ```

          // hooks/useBatchUpload.ts
          export function useBatchUpload(scope: MediaScope) {
            const [queue, setQueue] = useState&lt;UploadQueueItem[]&gt;([]);
            const [concurrency] = useState(3);

            async function addToQueue(files: File[], targetId: string) {
              const items = files.map(file => ({
                id: crypto.randomUUID(),
                file,
                targetId,
                status: 'queued' as const
              }));
              setQueue(prev => [...prev, ...items]);
              processQueue();
            }

            async function processQueue() {
              const pending = queue.filter(i => i.status === 'queued');
              const uploading = queue.filter(i => i.status === 'uploading');

              if (uploading.length >= concurrency || pending.length === 0) return;

              const next = pending.slice(0, concurrency - uploading.length);
              for (const item of next) {
                uploadItem(item);
              }
            }

            return {
              queue,
              addToQueue,
              cancelAll: () => setQueue([]),
              retryFailed: () => { /* retry logic */ }
            };
          }
